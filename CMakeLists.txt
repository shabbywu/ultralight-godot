cmake_minimum_required(VERSION 3.24)

project(ultralight-godot)

option(BUILD_TESTING "Build SDK regression tests" OFF)
include(CTest)
set(ULTRALIGHT_SDK_VERSION "2.0.0-beta.2" CACHE STRING "Ultralight SDK package version")
option(COPY_ULTRALIGHT_BINARY "Stage the selected SDK libraries beside each target" ON)
add_subdirectory(externals/ulbind17)
get_target_property(ULTRALIGHT_INCLUDE_DIR Ultralight::Ultralight INTERFACE_INCLUDE_DIRECTORIES)
get_filename_component(ULTRALIGHT_GODOT_SDK_ROOT "${ULTRALIGHT_INCLUDE_DIR}/.." ABSOLUTE)
file(READ "${ULTRALIGHT_GODOT_SDK_ROOT}/VERSION.txt" ULTRALIGHT_GODOT_SDK_VERSION)
string(STRIP "${ULTRALIGHT_GODOT_SDK_VERSION}" ULTRALIGHT_GODOT_SDK_VERSION)
if(NOT ULTRALIGHT_GODOT_SDK_VERSION STREQUAL "2.0.0-beta.2")
  message(FATAL_ERROR "This integration requires SDK 2.0.0-beta.2; found ${ULTRALIGHT_GODOT_SDK_VERSION}")
endif()

function(ultralight_godot_stage_runtime target appcore)
  if(COPY_ULTRALIGHT_BINARY)
    set(libraries UltralightCore WebCore Ultralight)
    if(appcore)
      list(APPEND libraries AppCore)
    endif()
    foreach(library IN LISTS libraries)
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "$<TARGET_FILE:Ultralight::${library}>" "$<TARGET_FILE_DIR:${target}>" VERBATIM)
    endforeach()
  endif()
  if(APPLE)
    if(appcore)
      set(runtime_path "@executable_path")
    else()
      set(runtime_path "@loader_path")
    endif()
  elseif(UNIX)
    set(runtime_path "$ORIGIN")
  endif()
  if(DEFINED runtime_path)
    set_target_properties(${target} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${runtime_path}")
  endif()
endfunction()

find_package(godot-cpp CONFIG REQUIRED)
add_subdirectory(ultralight-view)
add_subdirectory(mini-browser)
